class Point
{
    public int X { get; init; }
    public int Y  { get; init; }

    public Point()
    {
        
    }

    public Point(int x, int y)
    {
        X =x;
        Y =y;
    }

    public void Display()
    {
        Console.WriteLine($"[{X}, {Y}]");
    }
}
