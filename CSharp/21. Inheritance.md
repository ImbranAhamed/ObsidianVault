
Ability of a class to inherit the core functionality of a base class and extending it to theie needs.

Types:
- Is - a // relationship
-  has - as or Can - do // Composition


*Composition*

class Radio
{
    public void Power(bool isPower)
    {
        Console.WriteLine(isPower ? "Radio Turned On": "Radio Turned Off");
    }
}

class Car
{
    private readonly Radio radio;

    public Car()
    {
        radio = new Radio();
    }

    public void TurnOnRadio(bool onOff)
    {
        radio.Power(onOff);
    }
}
