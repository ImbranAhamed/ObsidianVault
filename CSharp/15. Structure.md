-  Can't inherit from structure or class
-  Can't serve as base for other to inherit
-  **Can implement interfaces**

struct Point
{
	public int x;
	public int y;

	public Point(int x, int y)
	{
			this.x = x;
			this.y = y;
	}

	public void Display()
	{
			Console.WriteLine($"x: {x}, y: {y}");
	}

	public void Increment()
	{
		x++;
		y++;
	}

	public void Decrement()
	{
			x--;
			y--;
	}
}


**Readonly struct**:


**readonly struct** Point
{
	public int x { get; }
	public int y { get; }

	public Point(int x, int y)
	{
			this.x = x;
			this.y = y;
	}

	public void Display()
	{
			Console.WriteLine($"x: {x}, y: {y}");
	}
}

struct Point
{
	public **readonly int** x;
	public **readonly int** y;

	public Point(int x, int y)
	{
			this.x = x;
			this.y = y;
	}

	public void Display()
	{
			Console.WriteLine($"x: {x}, y: {y}");
	}
}
